```python
def calculate_average(numbers):
    """Calculates and returns the average of a list of numbers.

    Returns 0 if the list is empty to avoid ZeroDivisionError.
    """
    # Check if the list is empty (an empty list evaluates to False in Python)
    if not numbers:
        return 0

    # Calculate average: total sum divided by the number of elements
    average = sum(numbers) / len(numbers)

    return average
```

```javascript
/**
 * Calculates and returns the average of an array of numbers.
 * Returns 0 if the array is empty.
 * 
 * @param {number[]} numbers - An array of numbers.
 * @returns {number} The calculated average.
 */
function calculateAverage(numbers) {
    // Check if the array is empty to prevent division by zero
    // In JavaScript, we check the .length property of the array
    if (!numbers || numbers.length === 0) {
        return 0;
    }

    // Calculate total sum using the reduce() method
    // 'sum' accumulates the running total, starting at 0
    const totalSum = numbers.reduce((sum, current) => sum + current, 0);

    // Calculate average: total sum divided by the number of elements
    const average = totalSum / numbers.length;

    return average;
}

// --- Example Function Calls ---

// 1. Normal array of numbers
const scores = [85, 90, 78, 92, 88];
const averageScore = calculateAverage(scores);

// 2. Empty array test
const emptyScores = [];
const emptyAverage = calculateAverage(emptyScores);

// Display the results using console.log and template literals
console.log(`Scores: [${scores.join(', ')}]`);
console.log(`Calculated Average: ${averageScore}`);

console.log(`\nScores: [${emptyScores.join(', ')}]`);
console.log(`Calculated Average (empty list): ${emptyAverage}`);
```

---

### Key Differences & Equivalents

1. **Naming Conventions:**
   * Python commonly uses `snake_case` (e.g., `calculate_average`, `average_score`).
   * JavaScript conventionally uses `camelCase` (e.g., `calculateAverage`, `averageScore`).

2. **Empty Check:**
   * Python: `if not numbers:` checks truthiness directly.
   * JavaScript: Arrays evaluate to `true` even when empty (`[]`), so we explicitly check `numbers.length === 0`.

3. **Summing Elements:**
   * Python has a built-in `sum()` function.
   * JavaScript uses `numbers.reduce((acc, curr) => acc + curr, 0)` to sum array values.

4. **Output Printing:**
   * Python uses `print(f"...")`.
   * JavaScript uses `console.log(\`...\`)` with template literals (backticks `` ` ``).
